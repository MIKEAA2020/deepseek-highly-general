    71→def kappa_V_indicator(kV_orig, delta_b, b_wt, tau=0.05):
    72→    """
    73→    Main κ_V definition (v10):  κ_V(g) = κ_V^orig(g) · 𝟙[Δb(g) > τ·b_wt]
    74→
    75→    Zeros out κ_V for KOs that do NOT reduce biomass by more than 5% of WT
    76→    (the standard essentiality threshold). This restricts κ_V to the
    77→    sub-population of KOs that actually impact viability, removing the
    78→    flux-rerouting noise that decouples κ_V from essentiality on denser
    79→    metabolic models (cf. E16 result on iML1515).
    80→    """
    81→    return kV_orig * (delta_b > tau * b_wt).astype(float)

    92→def _pearson(x, y):
    93→    if np.std(x) < 1e-15 or np.std(y) < 1e-15:
    94→        return 0.0, 1.0
    95→    return _safe_corr(np.corrcoef(x, y)[0, 1])
    96→
    97→
    98→def _spearman(x, y):
    99→    if np.std(x) < 1e-15 or np.std(y) < 1e-15:
   100→        return 0.0, 1.0
   101→    rx, ry = pd.Series(x).rank().values, pd.Series(y).rank().values
   102→    r = np.corrcoef(rx, ry)[0, 1]
   103→    return _safe_corr(r)
   104→

   106→def _safe_corr(r):
   107→    """Approximate two-sided p-value via Fisher transform."""
   108→    n = len(r) if hasattr(r, "__len__") else 0
   109→    if not isinstance(r, float):
   110→        r = float(r)
   111→    if abs(r) >= 0.9999:
   112→        return r, 0.0
   113→    # Approximate n if not given — caller should pass n; default 100
   114→    n_eff = max(30, 100)
   115→    z = 0.5 * np.log((1 + r) / (1 - r))
   116→    from scipy.stats import norm
   117→    p = 2 * (1 - norm.cdf(abs(z) * np.sqrt(n_eff - 3)))
   118→    return r, p

   121→def _roc_auc(y_true, score):
   122→    """ROC AUC via Mann-Whitney U. Returns 0.5 if undefined."""
   123→    y_true = np.asarray(y_true).astype(int)
   124→    score = np.asarray(score, dtype=float)
   125→    if len(np.unique(y_true)) < 2:
   126→        return 0.5
   127→    n1 = y_true.sum()
   128→    n0 = len(y_true) - n1
   129→    if n1 == 0 or n0 == 0:
   130→        return 0.5
   131→    order = np.argsort(-score)
   132→    ranks = pd.Series(score[order]).rank(method="average").values
   133→    # Rank is among full sample; sum of ranks of positives
   134→    sum_pos = ranks[y_true[order] == 1].sum() if (y_true[order] == 1).any() else 0.0
   135→    # Actually recompute properly
   136→    s = pd.Series(score).rank(method="average").values
   137→    sum_pos = s[y_true == 1].sum()
   138→    u = sum_pos - n1 * (n1 + 1) / 2
   139→    return u / (n1 * n0)

   142→def _mcc(y_true, y_pred):
   143→    """Matthews correlation coefficient."""
   144→    y_true = np.asarray(y_true).astype(int)
   145→    y_pred = np.asarray(y_pred).astype(int)
   146→    tp = int(((y_true == 1) & (y_pred == 1)).sum())
   147→    tn = int(((y_true == 0) & (y_pred == 0)).sum())
   148→    fp = int(((y_true == 0) & (y_pred == 1)).sum())
   149→    fn = int(((y_true == 1) & (y_pred == 0)).sum())
   150→    num = (tp * tn) - (fp * fn)
   151→    den = np.sqrt((tp + fp) * (tp + fn) * (tn + fp) * (tn + fn))
   152→    return float(num / den) if den > 0 else 0.0

   155→def _precision_at_k(y_true, score, k):
   156→    """Precision @ K: fraction of top-K scored items that are positives."""
   157→    y_true = np.asarray(y_true).astype(int)
   158→    score = np.asarray(score, dtype=float)
   159→    order = np.argsort(-score)
   160→    if k > len(y_true):
   161→        k = len(y_true)
   162→    if k == 0:
   163→        return 0.0
   164→    return float(y_true[order[:k]].sum()) / k

   167→def _held_out_logistic_auc(y_true, score, seed=20260830, train_frac=0.7):
   168→    """Held-out 70/30 logistic regression AUC.
   169→    Aligned with E17 protocol: class_weight='balanced', stratify=y,
   170→    random_state=20260830 (so cross-rebuild comparison is apples-to-apples).
   171→    """
   172→    from sklearn.linear_model import LogisticRegression
   173→    from sklearn.metrics import roc_auc_score
   174→    from sklearn.model_selection import train_test_split
   175→    y_true = np.asarray(y_true).astype(int)
   176→    score = np.asarray(score, dtype=float).reshape(-1, 1)
   177→    if len(np.unique(y_true)) < 2 or np.var(score.flatten()) == 0:
   178→        return 0.5
   179→    try:
   180→        X_tr, X_te, y_tr, y_te = train_test_split(
   181→            score, y_true, test_size=1.0 - train_frac,
   182→            random_state=seed, stratify=y_true)
   183→    except ValueError:
   184→        return 0.5
   185→    if len(np.unique(y_tr)) < 2 or len(np.unique(y_te)) < 2:
   186→        return 0.5
   187→    clf = LogisticRegression(class_weight='balanced', max_iter=200)
   188→    clf.fit(X_tr, y_tr)
   189→    try:
   190→        return float(roc_auc_score(y_te, clf.predict_proba(X_te)[:, 1]))
   191→    except ValueError:
   192→        return 0.5

   381→    indicator_T = {t: float(delta_b_per_T[t] > 0.05 * b_wt)
   382→                   for t in bio_per_T}
   383→    print(f"  b_wt (T1) = {b_wt:.4f}; threshold = 0.05 * b_wt = {0.05*b_wt:.4f}")
   384→    print(f"  Per-time indicator mask:")
   385→    for t in sorted(bio_per_T):
   386→        print(f"    {t}: biomass = {bio_per_T[t]:.4f}, Δb = {delta_b_per_T[t]:+.4f}, "
   387→              f"indicator = {int(indicator_T[t])}")
   388→    # Apply: κ_V_new(gene, t) = κ_V_orig(gene, t) · indicator(t)
   389→    df = df.copy()
   390→    df["indicator_T"] = df["time_point"].map(indicator_T)
   391→    df["kV_v10"] = df["kappa_V_predicted"] * df["indicator_T"]
   392→    # E10 metric: Pearson/Spearman r(κ_V(gene, t), |log2 FC(gene, t)|) over all 736 pairs
   393→    x_orig = df["kappa_V_predicted"].values.astype(float)
   394→    x_new = df["kV_v10"].values.astype(float)
   395→    y = df["abs_log2_FC"].values.astype(float)
   396→    r_orig, p_orig = _pearson(np.log10(np.where(x_orig > 0, x_orig, 1e-12)), y)
   397→    r_new, p_new = _pearson(np.log10(np.where(x_new > 0, x_new, 1e-12)), y)
   398→    rho_orig, _ = _spearman(x_orig, y)
   399→    rho_new, _ = _spearman(x_new, y)
   400→    print(f"\n  All gene-T pairs (n={len(df)}):")
   401→    print(f"    Pearson r(log10 κ_V, |log2 FC|)  orig = {r_orig:+.4f}  "
   402→          f"v10 = {r_new:+.4f}  Δ = {r_new-r_orig:+.4f}")
   403→    print(f"    Spearman ρ(κ_V, |log2 FC|)     orig = {rho_orig:+.4f}  "
   404→          f"v10 = {rho_new:+.4f}  Δ = {rho_new-rho_orig:+.4f}")
   405→    # Per-gene max κ_V vs max |log2 FC|
   406→    gene_max_orig = df.groupby("gene")["kappa_V_predicted"].max()
   407→    gene_max_new = df.groupby("gene")["kV_v10"].max()
   408→    gene_max_fc = df.groupby("gene")["abs_log2_FC"].max()
   409→    r_g_orig, _ = _pearson(np.log10(np.where(gene_max_orig.values > 0,
   410→                                            gene_max_orig.values, 1e-12)),
   411→                          gene_max_fc.values)
   412→    r_g_new, _ = _pearson(np.log10(np.where(gene_max_new.values > 0,
   413→                                           gene_max_new.values, 1e-12)),
   414→                         gene_max_fc.values)
   415→    rho_g_orig, _ = _spearman(gene_max_orig.values, gene_max_fc.values)
   416→    rho_g_new, _ = _spearman(gene_max_new.values, gene_max_fc.values)
   417→    print(f"\n  Per-gene max (n={len(gene_max_orig)} genes):")
   418→    print(f"    Pearson r(log10 max κ_V, max |log2 FC|)  orig = {r_g_orig:+.4f}  "
   419→          f"v10 = {r_g_new:+.4f}  Δ = {r_g_new-r_g_orig:+.4f}")
   420→    print(f"    Spearman ρ                            orig = {rho_g_orig:+.4f}  "
   421→          f"v10 = {rho_g_new:+.4f}  Δ = {rho_g_new-rho_g_orig:+.4f}")
   422→    # Top-5 genes by κ_V
   423→    top_orig = gene_max_orig.sort_values(ascending=False).head(5)
   424→    top_new = gene_max_new.sort_values(ascending=False).head(5)
   425→    print(f"\n  Top-5 genes by max κ_V:")
   426→    print(f"    ORIG: {', '.join(f'{g}({v:.4f})' for g, v in top_orig.items())}")
   427→    print(f"    V10:  {', '.join(f'{g}({v:.4f})' for g, v in top_new.items())}")

   434→        "biomass_per_T": bio_per_T,
   435→        "delta_b_per_T": delta_b_per_T,
   436→        "all_pairs": {"pearson_r": {"orig": r_orig, "v10": r_new},
   437→                      "spearman_rho": {"orig": rho_orig, "v10": rho_new}},

   439→                         "spearman_rho": {"orig": rho_g_orig, "v10": rho_g_new}},

   446→# 6. E15 + E16 re-run (cross-rebuild: iJO1366 vs iML1515)
   447→# ---------------------------------------------------------------------------
   448→def section_4_e15_e16():
   449→    print("\n" + "=" * 78)
   450→    print("E18 §4 — E15 + E16 RE-RUN (cross-rebuild, via E17 indicator)")
   451→    print("=" * 78)
   452→    # Load E15 (iJO1366) and E16 (iML1515)
   453→    df_ijo = pd.read_csv(f"{DOWNLOAD}/novelty_keio_direct_e15.csv")
   454→    df_iml = pd.read_csv(f"{DOWNLOAD}/novelty_keio_iml1515_e16.csv")
   455→    # Binary subset (drop u)
   456→    df_ijo = df_ijo[df_ijo["keio_call"].isin(["E", "N"])].copy().reset_index(drop=True)
   457→    df_iml = df_iml[df_iml["keio_call"].isin(["E", "N"])].copy().reset_index(drop=True)
   458→    y_ijo = (df_ijo["keio_call"] == "E").astype(int).values
   459→    y_iml = (df_iml["keio_call"] == "E").astype(int).values
   460→    b_wt_ijo = df_ijo["b_wt"].iloc[0]
   461→    b_wt_iml = df_iml["b_wt"].iloc[0]
   462→    kV_ijo = df_ijo["kV"].values.astype(float)
   463→    kV_iml = df_iml["kV"].values.astype(float)
   464→    db_ijo = df_ijo["delta_b"].values.astype(float)
   465→    db_iml = df_iml["delta_b"].values.astype(float)
   466→    # Apply indicator
   467→    kV_ijo_new = kV_ijo * (db_ijo > 0.05 * b_wt_ijo).astype(float)
   468→    kV_iml_new = kV_iml * (db_iml > 0.05 * b_wt_iml).astype(float)
   469→    # Metrics per model × variant
   470→    def metrics(y, kV):
   471→        log = _safe_log10(kV)
   472→        r, _ = _pearson(log, y)
   473→        rho, _ = _spearman(kV, y)
   474→        auc = _roc_auc(y, log)
   475→        held = _held_out_logistic_auc(y, log)
   476→        return r, rho, auc, held
   477→    r_o, rho_o, auc_o, held_o = metrics(y_ijo, kV_ijo)
   478→    r_n, rho_n, auc_n, held_n = metrics(y_ijo, kV_ijo_new)
   479→    r_o2, rho_o2, auc_o2, held_o2 = metrics(y_iml, kV_iml)
   480→    r_n2, rho_n2, auc_n2, held_n2 = metrics(y_iml, kV_iml_new)
   481→    print(f"  iJO1366 (n={len(df_ijo)}, b_wt={b_wt_ijo:.4f}, "
   482→          f"n_essential_raw_Keio={int(y_ijo.sum())})")
   483→    print(f"  iML1515 (n={len(df_iml)}, b_wt={b_wt_iml:.4f}, "
   484→          f"n_essential_raw_Keio={int(y_iml.sum())})")
   485→    print()
   486→    print(f"  {'metric':32s} {'iJO1366 orig':>13s} {'iJO1366 v10':>13s} "
   487→          f"{'iML1515 orig':>13s} {'iML1515 v10':>13s}")
   488→    print(f"  {'-'*84}")
   489→    print(f"  {'Pearson r(log10 κ_V, Keio-E)':32s} {r_o:>+13.4f} {r_n:>+13.4f} "
   490→          f"{r_o2:>+13.4f} {r_n2:>+13.4f}")
   491→    print(f"  {'Spearman ρ':32s} {rho_o:>+13.4f} {rho_n:>+13.4f} "
   492→          f"{rho_o2:>+13.4f} {rho_n2:>+13.4f}")
   493→    print(f"  {'ROC AUC':32s} {auc_o:>13.4f} {auc_n:>13.4f} "
   494→          f"{auc_o2:>13.4f} {auc_n2:>13.4f}")
   495→    print(f"  {'Held-out 70/30 AUC':32s} {held_o:>13.4f} {held_n:>13.4f} "
   496→          f"{held_o2:>13.4f} {held_n2:>13.4f}")
   497→    # Cross-rebuild stability |Δ|
   498→    def stab(o, n, oo, nn):
   499→        d_orig = abs(oo - o)
   500→        d_new = abs(nn - n)
   501→        return d_orig, d_new, d_new - d_orig
   502→    sp, sn, _ = stab(r_o, r_n, r_o2, r_n2)
   503→    sp2, sn2, _ = stab(rho_o, rho_n, rho_o2, rho_n2)
   504→    sp3, sn3, _ = stab(auc_o, auc_n, auc_o2, auc_n2)
   505→    sp4, sn4, _ = stab(held_o, held_n, held_o2, held_n2)
   506→    print()
   507→    print(f"  Cross-rebuild stability |Δ| (smaller is better):")
   508→    print(f"    {'metric':32s} {'orig |Δ|':>10s} {'v10 |Δ|':>10s} {'change':>10s}")
   509→    print(f"    {'-'*62}")
   510→    print(f"    {'Pearson r':32s} {sp:>10.4f} {sn:>10.4f} {sn-sp:>+10.4f}")
   511→    print(f"    {'Spearman ρ':32s} {sp2:>10.4f} {sn2:>10.4f} {sn2-sp2:>+10.4f}")
   512→    print(f"    {'ROC AUC':32s} {sp3:>10.4f} {sn3:>10.4f} {sn3-sp3:>+10.4f}")
   513→    print(f"    {'Held-out AUC':32s} {sp4:>10.4f} {sn4:>10.4f} {sn4-sp4:>+10.4f}")
   514→    return {
   515→        "iJO1366": {"n": int(len(df_ijo)), "b_wt": float(b_wt_ijo),
   516→                    "n_essential_raw_Keio": int(y_ijo.sum()),
   517→                    "orig": {"pearson_r": r_o, "spearman_rho": rho_o,
   518→                             "roc_auc": auc_o, "held_out_auc": held_o},
   519→                    "v10": {"pearson_r": r_n, "spearman_rho": rho_n,
   520→                            "roc_auc": auc_n, "held_out_auc": held_n}},
   521→        "iML1515": {"n": int(len(df_iml)), "b_wt": float(b_wt_iml),
   522→                    "n_essential_raw_Keio": int(y_iml.sum()),
   523→                    "orig": {"pearson_r": r_o2, "spearman_rho": rho_o2,
   524→                             "roc_auc": auc_o2, "held_out_auc": held_o2},
   525→                    "v10": {"pearson_r": r_n2, "spearman_rho": rho_n2,
   526→                            "roc_auc": auc_n2, "held_out_auc": held_n2}},
   527→        "stability": {"pearson_r": {"orig_abs_delta": sp, "v10_abs_delta": sn},
   528→                      "spearman_rho": {"orig_abs_delta": sp2,
   529→                                       "v10_abs_delta": sn2},
   530→                      "roc_auc": {"orig_abs_delta": sp3, "v10_abs_delta": sn3},
   531→                      "held_out_auc": {"orig_abs_delta": sp4,
   532→                                       "v10_abs_delta": sn4}},
   533→    }
