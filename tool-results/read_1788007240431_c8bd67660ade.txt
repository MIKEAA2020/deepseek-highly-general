   224→class CoverPage(Flowable):
   225→    def __init__(self):
   226→        super().__init__()
   227→        self.width = 0
   228→        self.height = 0
   229→    def draw(self):
   230→        pass
   231→
   232→
   233→class HorizontalRule(Flowable):
   234→    def __init__(self, width, thickness=0.5, color=C_ACCENT_A):
   235→        super().__init__()
   236→        self.width = width
   237→        self.thickness = thickness
   238→        self.color = color
   239→    def draw(self):
   240→        c = self.canv
   241→        c.setStrokeColor(self.color)
   242→        c.setLineWidth(self.thickness)
   243→        c.line(0, 0, self.width, 0)

   246→def part_divider(label, title, blurb):
   247→    return KeepTogether([
   248→        Spacer(1, 18),
   249→        Paragraph(label, style_part_label),
   250→        Paragraph(title, style_h1),
   251→        HRFlowable(width="100%", thickness=1.2, color=C_ACCENT_A, spaceBefore=2, spaceAfter=10),
   252→        Paragraph(blurb, style_body),
   253→        Spacer(1, 8),
   254→    ])

   257→def section(heading, paragraphs, quote=None, level=2):
   258→    style = style_h2 if level == 2 else (style_h3 if level == 3 else style_h3)
   259→    flow = [Paragraph(heading, style)]
   260→    if quote:
   261→        flow.append(Paragraph(quote, style_quote))
   262→    for p in paragraphs:
   263→        flow.append(Paragraph(p, style_body))
   264→    return flow
