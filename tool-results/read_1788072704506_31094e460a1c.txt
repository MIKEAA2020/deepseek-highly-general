   571→   571→def simulate_network(network, knockout_species=None, T=500, delta=0.05,
   572→   572→                     k_cat_metabolic=0.8, k_cat_synthesis=2.0,
   573→   573→                     k_cat_constitutive=0.3, k_cat_autocatalytic=1.0,
   574→   574→                     food_supply_rate=2.0, food_conc=10.0,
   575→   575→                     Km=0.1, max_conc=100.0,
   576→   576→                     init_concs=None, allow_constitutive=True):
   577→   577→    """Simulate the network dynamics for T steps.
   578→   578→
   579→   579→    Uses Michaelis-Menten saturation for each substrate to prevent
   580→   580→    runaway positive feedback: rate = k * [cat] * prod(s_i / (Km + s_i)).
   581→   581→    """
   582→   582→    sp = network["species"]
   583→   583→    fd = set(network["food"])
   584→   584→    rxs = network["reactions"]
   585→   585→
   586→   586→    x = {s: (food_conc if s in fd else 0.1) for s in sp}
   587→   587→    if init_concs:
   588→   588→        for s, v in init_concs.items():
   589→   589→            x[s] = v
   590→   590→
   591→   591→    knockout_reactions = set()
   592→   592→    if knockout_species is not None:
   593→   593→        for r in rxs:
   594→   594→            produced = [s for s, d in r["stoich"].items() if d > 0]
   595→   595→            if knockout_species in produced:
   596→   596→                knockout_reactions.add(r["id"])
   597→   597→
   598→   598→    dt = 0.05
   599→   599→    trajectory = [{s: x[s] for s in sp}]
   600→   600→    for step in range(T):
   601→   601→        rates = {}
   602→   602→        for r in rxs:
   603→   603→            if r["id"] in knockout_reactions:
   604→   604→                rates[r["id"]] = 0.0
   605→   605→                continue
   606→   606→            subs = [(s, abs(d)) for s, d in r["stoich"].items() if d < 0]
   607→   607→            if any(x[s] <= 0 for s, _ in subs):
   608→   608→                rates[r["id"]] = 0.0
   609→   609→                continue
   610→   610→            kind = r["kind"]
   611→   611→            cat = r["catalyst"]
   612→   612→            if kind == "constitutive":
   613→   613→                k = k_cat_constitutive
   614→   614→                rate = k
   615→   615→                for s, _ in subs:
   616→   616→                    rate *= x[s] / (Km + x[s])
   617→   617→            else:
   618→   618→                cat_conc = x.get(cat, 0.0) if cat is not None else 0.0
   619→   619→                if cat_conc <= 0:
   620→   620→                    rates[r["id"]] = 0.0
   621→   621→                    continue
   622→   622→                if kind == "synthesis":
   623→   623→                    k = k_cat_synthesis
   624→   624→                elif kind == "autocatalytic":
   625→   625→                    k = k_cat_autocatalytic
   626→   626→                else:
   627→   627→                    k = k_cat_metabolic
   628→   628→                if "k_cat_override" in r:
   629→   629→                    k = r["k_cat_override"]
   630→   630→                rate = k * cat_conc
   631→   631→                for s, _ in subs:
   632→   632→                    rate *= x[s] / (Km + x[s])
   633→   633→            rates[r["id"]] = rate
   634→   634→
   635→   635→        dx = {s: 0.0 for s in sp}
   636→   636→        for r in rxs:
   637→   637→            rate = rates[r["id"]]
   638→   638→            for s, d in r["stoich"].items():
   639→   639→                dx[s] += d * rate
   640→   640→        for s in sp:
   641→   641→            dx[s] -= delta * x[s]
   642→   642→            if s in fd:
   643→   643→                dx[s] += food_supply_rate * (food_conc - x[s]) * 0.5
   644→   644→        for s in sp:
   645→   645→            x[s] = max(0.0, min(max_conc, x[s] + dt * dx[s]))
   646→   646→        trajectory.append({s: x[s] for s in sp})
   647→   647→    return trajectory

   650→   650→def simulate_network_recover(network, init, T=500, delta=0.05,
   651→   651→                             k_cat_metabolic=0.8, k_cat_synthesis=2.0,
   652→   652→                             k_cat_constitutive=0.3, k_cat_autocatalytic=1.0,
   653→   653→                             food_conc=10.0, food_supply_rate=2.0,
   654→   654→                             Km=0.1, max_conc=100.0):
   655→   655→    """Simulate from a given initial condition (recovery test)."""
   656→   656→    sp = network["species"]
   657→   657→    fd = set(network["food"])
   658→   658→    rxs = network["reactions"]
   659→   659→    x = {s: init.get(s, 0.0) for s in sp}
   660→   660→    for s in fd:
   661→   661→        x[s] = max(x[s], food_conc)
   662→   662→    if x.get("ATP", 0) < 1.0:
   663→   663→        x["ATP"] = max(x["ATP"], 2.0)
   664→   664→    if x.get("TF", 0) < 0.05:
   665→   665→        x["TF"] = max(x["TF"], 0.1)
   666→   666→    dt = 0.05
   667→   667→    trajectory = [{s: x[s] for s in sp}]
   668→   668→    for step in range(T):
   669→   669→        rates = {}
   670→   670→        for r in rxs:
   671→   671→            subs = [(s, abs(d)) for s, d in r["stoich"].items() if d < 0]
   672→   672→            if any(x[s] <= 0 for s, _ in subs):
   673→   673→                rates[r["id"]] = 0.0
   674→   674→                continue
   675→   675→            kind = r["kind"]
   676→   676→            cat = r["catalyst"]
   677→   677→            if kind == "constitutive":
   678→   678→                k = k_cat_constitutive
   679→   679→                rate = k
   680→   680→                for s, _ in subs:
   681→   681→                    rate *= x[s] / (Km + x[s])
   682→   682→            else:
   683→   683→                cat_conc = x.get(cat, 0.0) if cat is not None else 0.0
   684→   684→                if cat_conc <= 0:
   685→   685→                    rates[r["id"]] = 0.0
   686→   686→                    continue
   687→   687→                if kind == "synthesis":
   688→   688→                    k = k_cat_synthesis
   689→   689→                elif kind == "autocatalytic":
   690→   690→                    k = k_cat_autocatalytic
   691→   691→                else:
   692→   692→                    k = k_cat_metabolic
   693→   693→                if "k_cat_override" in r:
   694→   694→                    k = r["k_cat_override"]
   695→   695→                rate = k * cat_conc
   696→   696→                for s, _ in subs:
   697→   697→                    rate *= x[s] / (Km + x[s])
   698→   698→            rates[r["id"]] = rate
   699→   699→        dx = {s: 0.0 for s in sp}
   700→   700→        for r in rxs:
   701→   701→            rate = rates[r["id"]]
   702→   702→            for s, d in r["stoich"].items():
   703→   703→                dx[s] += d * rate
   704→   704→        for s in sp:
   705→   705→            dx[s] -= delta * x[s]
   706→   706→            if s in fd:
   707→   707→                dx[s] += food_supply_rate * (food_conc - x[s]) * 0.5
   708→   708→        for s in sp:
   709→   709→            x[s] = max(0.0, min(max_conc, x[s] + dt * dx[s]))
   710→   710→        trajectory.append({s: x[s] for s in sp})
   711→   711→    return trajectory

   714→   714→def closure_test(network, network_name, T=500, viability_threshold=0.1):
   715→   715→    """Run the autopoiesis closure test of Definition def:autopoiesis."""
   716→   716→    baseline = simulate_network(network, knockout_species=None, T=T)
   717→   717→    baseline_final = baseline[-1]
   718→   718→    records = []
   719→   719→    for m_j in network["non_food"]:
   720→   720→        knock = simulate_network(network, knockout_species=m_j, T=T)
   721→   721→        knock_final = knock[-1]
   722→   722→        knock_traj = [t[m_j] for t in knock]
   723→   723→        recover_start_idx = T // 2
   724→   724→        recover_init = knock[recover_start_idx]
   725→   725→        recover = simulate_network_recover(network, init=recover_init, T=T - recover_start_idx)
   726→   726→        recover_final = recover[-1]
   727→   727→        knock_success = knock_final[m_j] < viability_threshold
   728→   728→        recover_success = recover_final[m_j] > viability_threshold
   729→   729→        causally_internal = knock_success and recover_success
   730→   730→        records.append({
   731→   731→            "network": network_name,
   732→   732→            "component": m_j,
   733→   733→            "baseline_conc": baseline_final[m_j],
   734→   734→            "knockout_conc_final": knock_final[m_j],
   735→   735→            "knockout_min": min(knock_traj),
   736→   736→            "recover_conc_final": recover_final[m_j],
   737→   737→            "knockout_success": knock_success,
   738→   738→            "recover_success": recover_success,
   739→   739→            "causally_internal": causally_internal,
   740→   740→            "verdict": "AUTOPOIETIC" if causally_internal else "HOMEOSTATIC",
   741→   741→        })
   742→   742→    return records, baseline

   749→   749→def phase_iii_verdict(network, m_j, T=500, viability_threshold=0.1,
   750→   750→                      pathwise_fraction=0.4, perturbation_sigma=0.05,
   751→   751→                      contractibility_rel_tol=0.30, n_perturbed=2,
   752→   752→                      seed=20260831):
   753→   753→    """Phase III closure-test verdict for component m_j."""
   754→   754→    knock = simulate_network(network, knockout_species=m_j, T=T)
   755→   755→    knock_final = knock[-1]
   756→   756→    recover_start_idx = T // 2
   757→   757→    recover_init = knock[recover_start_idx]
   758→   758→    recover = simulate_network_recover(network, init=recover_init,
   759→   759→                                        T=T - recover_start_idx)
   760→   760→    recover_final = recover[-1]
   761→   761→    knock_success = knock_final[m_j] < viability_threshold
   762→   762→    recover_success_endpoint = recover_final[m_j] > viability_threshold
   763→   763→    phase_i_pass = knock_success and recover_success_endpoint
   764→   764→
   765→   765→    recovery_traj = [t[m_j] for t in recover]
   766→   766→    recovery_traj_mean = float(np.mean(recovery_traj))
   767→   767→    recovery_traj_max = float(np.max(recovery_traj))
   768→   768→    recovery_traj_min = float(np.min(recovery_traj))
   769→   769→    recovery_traj_above_frac = float(
   770→   770→        sum(1 for v in recovery_traj if v > viability_threshold)
   771→   771→        / max(1, len(recovery_traj))
   772→   772→    )
   773→   773→    pathwise_pass = recovery_traj_above_frac >= pathwise_fraction
   774→   774→
   775→   775→    rng = np.random.default_rng(seed)
   776→   776→    contractible_pass = True
   777→   777→    if pathwise_pass:
   778→   778→        for k in range(n_perturbed):
   779→   779→            perturbed_init = dict(recover_init)
   780→   780→            related = set()
   781→   781→            for r in network["reactions"]:
   782→   782→                if m_j in r["stoich"]:
   783→   783→                    for s in r["stoich"]:
   784→   784→                        if s != m_j and s not in network["food"]:
   785→   785→                            related.add(s)
   786→   786→            for s in [m_j] + list(related)[:5]:
   787→   787→                if s in perturbed_init:
   788→   788→                    noise = rng.normal(0, perturbation_sigma)
   789→   789→                    perturbed_init[s] = max(0.0, perturbed_init[s] * (1 + noise))
   790→   790→            rec_pert = simulate_network_recover(network, init=perturbed_init,
   791→   791→                                                 T=T - recover_start_idx)
   792→   792→            rec_pert_traj = [t[m_j] for t in rec_pert]
   793→   793→            pert_mean = float(np.mean(rec_pert_traj))
   794→   794→            pert_max = float(np.max(rec_pert_traj))
   795→   795→            pert_min = float(np.min(rec_pert_traj))
   796→   796→            for orig, pert in [(recovery_traj_mean, pert_mean),
   797→   797→                                (recovery_traj_max, pert_max),
   798→   798→                                (recovery_traj_min, pert_min)]:
   799→   799→                ref = max(abs(orig), abs(pert), 1e-3)
   800→   800→                if abs(orig - pert) / ref > contractibility_rel_tol:
   801→   801→                    contractible_pass = False
   802→   802→                    break
   803→   803→            if not contractible_pass:
   804→   804→                break
   805→   805→    else:
   806→   806→        contractible_pass = False
   807→   807→
   808→   808→    phase_iii_pass = phase_i_pass or (pathwise_pass and contractible_pass)
   809→   809→    return {
   810→   810→        "phase_i_pass": phase_i_pass,
   811→   811→        "pathwise_pass": pathwise_pass,
   812→   812→        "contractible_pass": contractible_pass,
   813→   813→        "phase_iii_pass": phase_iii_pass,
   814→   814→        "recovery_traj_mean": recovery_traj_mean,
   815→   815→        "recovery_traj_max": recovery_traj_max,
   816→   816→        "recovery_traj_min": recovery_traj_min,
   817→   817→        "recovery_traj_above_frac": recovery_traj_above_frac,
   818→   818→    }
