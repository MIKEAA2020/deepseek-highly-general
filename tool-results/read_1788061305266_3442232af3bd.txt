   236→   236→def stratified_holonomy_numerical(x_c, y_c, eps, F, alpha_fn, n_steps=4001):
   237→   237→    """Compute the stratified holonomy by parallel transport along the loop,
   238→   238→    using the connection 1-form A = (F/2)(x dy - y dx) on each stratum and
   239→   239→    the boundary transition g_{+-}(x, 0) = exp(i alpha(x)) on the boundary.
   240→   240→
   241→   241→    Uses n_steps = ODD number to ensure no edge sample lands exactly on y=0
   242→   242→    (which would defeat the strict-sign-change crossing detection).
   243→   243→    """
   244→   244→    # Ensure n_steps is odd so no sample point lands exactly on y=0
   245→   245→    if n_steps % 2 == 0:
   246→   246→        n_steps += 1
   247→   247→    # Corners of the rectangle (counterclockwise starting from bottom-left)
   248→   248→    corners = [
   249→   249→        (x_c - eps / 2, y_c - eps / 2),  # bottom-left (in S_- if y_c < eps/2)
   250→   250→        (x_c + eps / 2, y_c - eps / 2),  # bottom-right
   251→   251→        (x_c + eps / 2, y_c + eps / 2),  # top-right
   252→   252→        (x_c - eps / 2, y_c + eps / 2),  # top-left
   253→   253→    ]
   254→   254→    # Edges (each edge is a list of (x, y) points sampled)
   255→   255→    edges = []
   256→   256→    for i in range(4):
   257→   257→        start = corners[i]
   258→   258→        end = corners[(i + 1) % 4]
   259→   259→        ts = np.linspace(0, 1, n_steps + 1)
   260→   260→        pts = [(start[0] + t * (end[0] - start[0]),
   261→   261→                start[1] + t * (end[1] - start[1])) for t in ts]
   262→   262→        edges.append(pts)
   263→   263→    # Parallel transport: accumulate exp(-i * integral A) along each edge,
   264→   264→    # then apply boundary transitions at crossings.
   265→   265→    h = 1.0 + 0j  # accumulated holonomy (start at identity)
   266→   266→    n_crossings = 0
   267→   267→    for edge_idx, edge in enumerate(edges):
   268→   268→        # Walk along the edge, accumulating the holonomy
   269→   269→        # detect boundary crossings (y = 0)
   270→   270→        for i in range(len(edge) - 1):
   271→   271→            x1, y1 = edge[i]
   272→   272→            x2, y2 = edge[i + 1]
   273→   273→            # check if y crosses 0 between (x1, y1) and (x2, y2)
   274→   274→            # (strict sign change, since n_steps is odd no sample lands on y=0)
   275→   275→            if y1 * y2 < 0:
   276→   276→                n_crossings += 1
   277→   277→                # find crossing point
   278→   278→                t_cross = (0 - y1) / (y2 - y1)
   279→   279→                x_cross = x1 + t_cross * (x2 - x1)
   280→   280→                # accumulate holonomy from edge[i] to crossing point
   281→   281→                # (in the stratum that edge[i] is in)
   282→   282→                # then apply boundary transition g_{+-}(x_cross) = exp(i alpha(x_cross))
   283→   283→                # then continue in the other stratum
   284→   284→                # for abelian U(1), order doesn't matter; we accumulate the
   285→   285→                # stratum holonomy up to the crossing, then multiply by the
   286→   286→                # boundary transition
   287→   287→                # stratum segment (x1, y1) -> (x_cross, 0)
   288→   288→                # integrate A over this segment
   289→   289→                dx = x_cross - x1
   290→   290→                dy = 0 - y1
   291→   291→                # midpoint for trapezoidal rule
   292→   292→                x_mid = (x1 + x_cross) / 2
   293→   293→                y_mid = (y1 + 0) / 2
   294→   294→                A_val = (F / 2) * (x_mid * dy - y_mid * dx)
   295→   295→                h *= np.exp(-1j * A_val)
   296→   296→                # apply boundary transition (going from stratum of y1 to stratum of y2)
   297→   297→                # if y1 > 0 (S_+) and y2 < 0 (S_-): apply g_{+-}(x_cross) = exp(i alpha(x_cross))
   298→   298→                # if y1 < 0 (S_-) and y2 > 0 (S_+): apply g_{-+}(x_cross) = exp(-i alpha(x_cross))
   299→   299→                if y1 > 0:
   300→   300→                    h *= np.exp(1j * alpha_fn(x_cross))
   301→   301→                else:
   302→   302→                    h *= np.exp(-1j * alpha_fn(x_cross))
   303→   303→                # accumulate stratum holonomy from crossing to edge[i+1]
   304→   304→                dx = x2 - x_cross
   305→   305→                dy = y2 - 0
   306→   306→                x_mid = (x_cross + x2) / 2
   307→   307→                y_mid = (0 + y2) / 2
   308→   308→                A_val = (F / 2) * (x_mid * dy - y_mid * dx)
   309→   309→                h *= np.exp(-1j * A_val)
   310→   310→            else:
   311→   311→                # no crossing: accumulate stratum holonomy
   312→   312→                dx = x2 - x1
   313→   313→                dy = y2 - y1
   314→   314→                x_mid = (x1 + x2) / 2
   315→   315→                y_mid = (y1 + y2) / 2
   316→   316→                A_val = (F / 2) * (x_mid * dy - y_mid * dx)
   317→   317→                h *= np.exp(-1j * A_val)
   318→   318→    return h, np.angle(h), n_crossings
