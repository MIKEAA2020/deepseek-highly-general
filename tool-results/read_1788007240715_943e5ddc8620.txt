   179→class CoverPage(Flowable):
   180→    """A zero-size flowable that simply triggers the cover, then page break."""
   181→    def __init__(self):
   182→        super().__init__()
   183→        self.width = 0
   184→        self.height = 0
   185→
   186→    def draw(self):
   187→        pass  # actual cover drawn via onFirstPage callback
   188→
   189→
   190→class HorizontalRule(Flowable):
   191→    def __init__(self, width, thickness=0.5, color=C_ACCENT):
   192→        super().__init__()
   193→        self.width = width
   194→        self.thickness = thickness
   195→        self.color = color
   196→
   197→    def draw(self):
   198→        c = self.canv
   199→        c.setStrokeColor(self.color)
   200→        c.setLineWidth(self.thickness)
   201→        c.line(0, 0, self.width, 0)

   204→def part_divider(label, title, blurb):
   205→    """A section divider Flowable that introduces a Part."""
   206→    return KeepTogether([
   207→        Spacer(1, 18),
   208→        Paragraph(label, style_part_label),
   209→        Paragraph(title, style_h1),
   210→        HRFlowable(width="100%", thickness=1.2, color=C_ACCENT, spaceBefore=2, spaceAfter=10),
   211→        Paragraph(blurb, style_body),
   212→        Spacer(1, 8),
   213→    ])
   214→
   215→
   216→def section(heading, paragraphs, quote=None):
   217→    """A standard section: heading + optional quote + body paragraphs."""
   218→    flow = [Paragraph(heading, style_h2)]
   219→    if quote:
   220→        flow.append(Paragraph(quote, style_quote))
   221→    for p in paragraphs:
   222→        flow.append(Paragraph(p, style_body))
   223→    return flow
