   170→def section_heading(title):
   171→    """Inline section heading (no full-page divider)."""
   172→    return KeepTogether([
   173→        Spacer(1, 4),
   174→        Paragraph(title, style_h1),
   175→        HRFlowable(width="100%", thickness=0.8, color=C_ACCENT,
   176→                   spaceBefore=1, spaceAfter=4),
   177→    ])

   180→def claim_block(claim_label, claim_text, method_text, evidence_text, implication_text):
   181→    """Render a structured claim block: Claim / Method / Evidence / Implication."""
   182→    out = []
   183→    out.append(Paragraph(claim_label, style_h3))
   184→    out.append(Paragraph(f"<b>Claim.</b> {claim_text}", style_body))
   185→    out.append(Paragraph(f"<b>Method.</b> {method_text}", style_body))
   186→    out.append(Paragraph(f"<b>Evidence.</b> {evidence_text}", style_body))
   187→    out.append(Paragraph(f"<b>Implication.</b> {implication_text}", style_body))
   188→    out.append(Spacer(1, 2))
   189→    return out
