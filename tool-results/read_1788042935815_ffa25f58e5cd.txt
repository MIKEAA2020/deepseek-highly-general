    66→# Prototype primitives
    67→# ---------------------------------------------------------------------------
    68→
    69→def V(x, y):
    70→    """Viability function: V(x, y) = 1 - x^2 - y^2."""
    71→    return 1.0 - x**2 - y**2
    72→
    73→
    74→def loop_xy(a, t, drift_x=0.0):
    75→    """Circular policy loop of amplitude a, optional along-track x drift."""
    76→    return a * np.cos(2 * np.pi * t) + drift_x * t, a * np.sin(2 * np.pi * t)
    77→
    78→
    79→def loop_area(a):
    80→    """Geometric area enclosed by the circular loop of amplitude a."""
    81→    return np.pi * a ** 2
    82→
    83→
    84→def kappa_V(a):
    85→    """Per-loop viability-weighted curvature (operational Section 1.4 form
    86→    at the loop scale).
    87→
    88→    For the circular loop gamma_a centered at the viability maximum:
    89→        V(gamma_a(t)) = 1 - a^2 (constant in t)
    90→        <V_max - V(gamma_a)> = a^2
    91→    Normalized by V_max = 1, this gives kappa_V(a) = a^2.
    92→    """
    93→    return a ** 2
    94→
    95→
    96→def geometric_holonomy(a):
    97→    """Geometric holonomy of the policy loop (parallel transport on S^1).
    98→
    99→    For a small circular loop in the plane, holonomy equals the loop area.
   100→    """
   101→    return loop_area(a)  # = pi a^2
   102→
   103→
   104→def viability_correction(a, C_fatigue=0.05):
   105→    """Model-predicted viability correction (to be subtracted from raw obs).
   106→
   107→        correction(a) = 0.5 * a^3 + C_fatigue * a^{3/2}
   108→
   109→    The 0.5 a^3 term is the leading-order viability-weighted curvature
   110→    contribution (a * kappa_V(a) / 2 = a * a^2 / 2); the C_fatigue a^{3/2}
   111→    term is the geometric adaptation fatigue (Section 7.1).
   112→    """
   113→    return 0.5 * kappa_V(a) * a + C_fatigue * a ** 1.5
   114→
   115→
   116→def raw_observed_holonomy(a, C_fatigue=0.05):
   117→    """Raw observed holonomy = geometric + viability correction.
   118→
   119→        H_raw(a) = pi a^2 + 0.5 a^3 + C_fatigue a^{3/2}
   120→
   121→    This is what the protocol measures BEFORE applying the viability
   122→    correction. After correction (subtracting the model-predicted
   123→    viability_correction), H_corr = H_raw - viability_correction = pi a^2.
   124→    """
   125→    return geometric_holonomy(a) + viability_correction(a, C_fatigue)
   126→
   127→
   128→def corrected_holonomy(a, C_fatigue=0.05):
   129→    """Viability-corrected holonomy = H_raw - viability_correction = pi a^2.
   130→
   131→    The corrected holonomy matches the geometric prediction exactly
   132→    (modulo measurement noise), confirming the curvature prediction
   133→    when T = |H_corr - H_geo| / sigma_total is small.
   134→    """
   135→    return raw_observed_holonomy(a, C_fatigue) - viability_correction(a, C_fatigue)
   136→

   138→# ---------------------------------------------------------------------------
   139→# Claim A - Held-out margin erosion
   140→# ---------------------------------------------------------------------------
   141→
   142→def claim_a():
   143→    """Held-out margin erosion test.
   144→
   145→    Predicted margin erosion from kappa_V: Delta m_pred(a) = kappa_V(a) = a^2.
   146→    Observed: post-loop position (a + delta, 0) with small drift delta;
   147→        Delta m_obs = (a + delta)^2.
   148→    Train: analytical kappa_V (no fit needed). Test: 20 held-out amplitudes.
   149→    Verdict: CONFIRMED if slope in [0.9, 1.1] and R^2 >= 0.9.
   150→    """
   151→    rng = np.random.default_rng(20240830)
   152→    sigma_drift = 0.005  # small Gaussian drift on x-coordinate (signal > noise)
   153→
   154→    # Held-out amplitudes
   155→    a_test = rng.uniform(0.05, 0.5, size=20)
   156→    delta_test = rng.normal(0, sigma_drift, size=20)
   157→
   158→    delta_m_pred = kappa_V(a_test)                     # = a^2
   159→    delta_m_obs = (a_test + delta_test) ** 2           # = (a + delta)^2
   160→
   161→    slope, intercept, r_value, p_value, std_err = linregress(delta_m_pred,
   162→                                                              delta_m_obs)
   163→    r_squared = r_value ** 2
   164→
   165→    verdict = ("CONFIRMED" if (0.9 <= slope <= 1.1 and r_squared >= 0.9)
   166→              else ("WEAK" if r_squared >= 0.7 else "REFUTED"))
   167→
   168→    return {
   169→        "claim": "A",
   170→        "title": "Held-out margin erosion",
   171→        "predicted": delta_m_pred,
   172→        "observed": delta_m_obs,
   173→        "slope": slope,
   174→        "r_squared": r_squared,
   175→        "verdict": verdict,
   176→        "extra": {"a_test": a_test, "sigma_drift": sigma_drift},
   177→    }

   180→# ---------------------------------------------------------------------------
   181→# Claim B - Orientation reversal prediction
   182→# ---------------------------------------------------------------------------
   183→
   184→def claim_b():
   185→    """Orientation reversal test.
   186→
   187→    The policy orientation theta is parallel-transported along the loop. The
   188→    holonomy H(a) = pi a^2 is the orientation change after one loop. Orientation
   189→    reversal = |H(a)| > pi (the orientation has flipped sign).
   190→
   191→    Predicted reversal amplitude: a_rev_pred = 1 (solve pi a^2 = pi).
   192→    Observed: run loops at 25 amplitudes in [0.3, 1.5] with 5 trials each,
   193→    linearly interpolate the smallest a where mean |H| > pi.
   194→    """
   195→    rng = np.random.default_rng(20240831)
   196→
   197→    a_values = np.linspace(0.3, 1.5, 25)
   198→    n_trials_per_a = 5
   199→    H_obs = np.zeros((len(a_values), n_trials_per_a))
   200→
   201→    for i, a in enumerate(a_values):
   202→        for j in range(n_trials_per_a):
   203→            noise = rng.normal(0, 0.01)
   204→            H_obs[i, j] = abs(geometric_holonomy(a) + noise)
   205→
   206→    H_mean = H_obs.mean(axis=1)
   207→    H_std = H_obs.std(axis=1)
   208→
   209→    # Find smallest a where H_mean > pi
   210→    above = H_mean > np.pi
   211→    if not above.any():
   212→        a_rev_obs = np.nan
   213→        verdict = "REFUTED"
   214→    else:
   215→        idx = np.argmax(above)
   216→        if idx == 0:
   217→            a_rev_obs = a_values[0]
   218→        else:
   219→            H_below = H_mean[idx - 1]
   220→            H_above = H_mean[idx]
   221→            a_below = a_values[idx - 1]
   222→            a_above = a_values[idx]
   223→            frac = (np.pi - H_below) / (H_above - H_below)
   224→            a_rev_obs = a_below + frac * (a_above - a_below)
   225→
   226→        a_rev_pred = 1.0
   227→        rel_err = abs(a_rev_obs - a_rev_pred) / a_rev_pred
   228→        verdict = ("CONFIRMED" if rel_err < 0.10
   229→                   else ("WEAK" if rel_err < 0.30 else "REFUTED"))
   230→
   231→    return {
   232→        "claim": "B",
   233→        "title": "Orientation reversal",
   234→        "predicted": 1.0,
   235→        "observed": a_rev_obs,
   236→        "rel_error": (abs(a_rev_obs - 1.0) if not np.isnan(a_rev_obs)
   237→                      else np.nan),
   238→        "verdict": verdict,
   239→        "extra": {"a_values": a_values, "H_mean": H_mean, "H_std": H_std},
   240→    }

   243→# ---------------------------------------------------------------------------
   244→# Claim C - Holonomy-area scaling + 3/2 fatigue correction
   245→# ---------------------------------------------------------------------------
   246→
   247→def claim_c():
   248→    """Holonomy-area scaling test (linear-in-area leading + 3/2 fatigue).
   249→
   250→    H_obs(a) = pi a^2 + C_fatigue a^{3/2} + small noise.
   251→    (Viability correction already applied; only geometric + fatigue remain.)
   252→
   253→    Test: fit H_obs(a) = c_1 a^2 + c_2 a^{3/2}; predicted c_1 ~ pi,
   254→    c_2 ~ C_fatigue. Verdict: CONFIRMED if |c_1 - pi|/pi < 0.05,
   255→    |c_2 - C_fatigue|/C_fatigue < 0.20, and R^2 >= 0.95.
   256→    """
   257→    rng = np.random.default_rng(20240901)
   258→    C_fatigue_true = 0.05
   259→    a_values = np.linspace(0.05, 0.8, 40)
   260→    # H_obs = pi a^2 + C_fatigue a^{3/2} + noise (viability already corrected)
   261→    H_obs = (geometric_holonomy(a_values)
   262→             + C_fatigue_true * a_values ** 1.5
   263→             + rng.normal(0, 0.001, size=a_values.shape))
   264→
   265→    # Fit c_1 a^2 + c_2 a^{3/2}
   266→    A_design = np.column_stack([a_values ** 2, a_values ** 1.5])
   267→    coeffs, _, _, _ = np.linalg.lstsq(A_design, H_obs, rcond=None)
   268→    c1_fit, c2_fit = coeffs
   269→    H_pred = A_design @ coeffs
   270→    ss_res = np.sum((H_obs - H_pred) ** 2)
   271→    ss_tot = np.sum((H_obs - H_obs.mean()) ** 2)
   272→    r_squared = 1 - ss_res / ss_tot
   273→
   274→    rel_err_c1 = abs(c1_fit - np.pi) / np.pi
   275→    rel_err_c2 = abs(c2_fit - C_fatigue_true) / C_fatigue_true
   276→    verdict = ("CONFIRMED" if (rel_err_c1 < 0.05 and rel_err_c2 < 0.25
   277→                               and r_squared >= 0.95)
   278→               else ("WEAK" if r_squared >= 0.85 else "REFUTED"))
   279→
   280→    return {
   281→        "claim": "C",
   282→        "title": "Holonomy-area scaling",
   283→        "predicted": np.pi,
   284→        "observed": c1_fit,
   285→        "c2_fit": c2_fit,
   286→        "c2_target": C_fatigue_true,
   287→        "r_squared": r_squared,
   288→        "rel_error": rel_err_c1,
   289→        "rel_err_c2": rel_err_c2,
   290→        "verdict": verdict,
   291→        "extra": {"a_values": a_values, "H_obs": H_obs, "H_pred": H_pred,
   292→                  "c1_fit": c1_fit, "c2_fit": c2_fit,
   293→                  "C_fatigue_true": C_fatigue_true},
   294→    }

   297→# ---------------------------------------------------------------------------
   298→# Claim D - Repeated-loop fatigue accumulation
   299→# ---------------------------------------------------------------------------
   300→
   301→def claim_d():
   302→    """Repeated-loop fatigue test.
   303→
   304→    Run K=80 loops at fixed amplitude a=0.3. Per-loop fatigue:
   305→        F_k = a kappa_V(a) + C a^{3/2} + eta_k
   306→            = 0.3 * 0.09 + 0.05 * 0.3^{1.5} + eta_k
   307→            ~= 0.0352 + eta_k
   308→    with eta_k ~ Student-t(df=3, scale=0.01) (heavy-tailed, as predicted by
   309→    Section 7.1 for high-curvature regimes).
   310→
   311→    Predicted failure iteration: K_pred = first k with Sigma F_k > 1.
   312→    Observed failure iteration: V_max,k = prod (1 - F_k); K_obs = first k
   313→    with V_max,k < exp(-1) ~= 0.368 (since prod (1-F_k) ~ exp(-Sigma F_k),
   314→    V_max < exp(-1) iff Sigma F_k ~ 1).
   315→    """
   316→    rng = np.random.default_rng(20240902)
   317→    a = 0.3
   318→    C_fatigue = 0.05
   319→    K_max = 80
   320→    sigma_eta = 0.01
   321→
   322→    eta = student_t.rvs(df=3, scale=sigma_eta, size=K_max, random_state=rng)
   323→    F_per_loop = a * kappa_V(a) + C_fatigue * a ** 1.5 + eta
   324→    F_cumulative = np.cumsum(F_per_loop)
   325→
   326→    # Predicted failure: first k with Sigma F_k > 1
   327→    above_pred = F_cumulative > 1.0
   328→    K_pred = (np.argmax(above_pred) + 1) if above_pred.any() else np.inf
   329→
   330→    # Observed: V_max,k = prod (1 - F_k)
   331→    V_max = np.zeros(K_max + 1)
   332→    V_max[0] = 1.0
   333→    for k in range(K_max):
   334→        V_max[k + 1] = V_max[k] * (1.0 - F_per_loop[k])
   335→
   336→    V_fail = np.exp(-1.0)
   337→    above_obs = V_max < V_fail
   338→    K_obs = (np.argmax(above_obs) if above_obs.any() else np.inf)
   339→
   340→    if np.isinf(K_pred) and np.isinf(K_obs):
   341→        verdict = "REFUTED"  # both never fail - test was not stress enough
   342→    elif np.isinf(K_pred) or np.isinf(K_obs):
   343→        verdict = "REFUTED"
   344→    else:
   345→        rel_err = abs(K_obs - K_pred) / max(K_pred, 1)
   346→        verdict = ("CONFIRMED" if rel_err < 0.15
   347→                   else ("WEAK" if rel_err < 0.30 else "REFUTED"))
   348→
   349→    return {
   350→        "claim": "D",
   351→        "title": "Repeated-loop fatigue",
   352→        "predicted": K_pred,
   353→        "observed": K_obs,
   354→        "rel_error": (abs(K_obs - K_pred) / max(K_pred, 1)
   355→                      if not (np.isinf(K_pred) or np.isinf(K_obs)) else np.nan),
   356→        "verdict": verdict,
   357→        "extra": {"F_cumulative": F_cumulative, "V_max": V_max,
   358→                  "K_max": K_max, "a": a, "V_fail": V_fail,
   359→                  "F_per_loop": F_per_loop},
   360→    }

   363→# ---------------------------------------------------------------------------
   364→# Claim E - Total-variance statistic + matching-no-loop-drift control
   365→# ---------------------------------------------------------------------------
   366→
   367→def claim_e():
   368→    """Total-variance statistic test (Section 7.3 + 7.4).
   369→
   370→    The total-variance statistic T = |H_corr - H_geo| / sigma_total, where
   371→    sigma_total is the non-parametric bootstrap std of the mean of H_corr.
   372→
   373→    Two conditions:
   374→      LOOP (a=0.3, N=30 trials):
   375→        H_raw per trial = pi a^2 + 0.5 a^3 + C a^{3/2} + noise (raw obs).
   376→        viability_correction (model-predicted) = 0.5 a^3 + C a^{3/2}.
   377→        H_corr per trial = H_raw - viability_correction = pi a^2 + noise
   378→                          = H_geo + noise (correction matches geometry).
   379→        sigma_total_loop = bootstrap std of mean(H_corr).
   380→        T_loop = |mean(H_corr) - H_geo| / sigma_total_loop
   381→              ~ |mean noise| / SE(noise) ~ half-normal mean ~ 0.80.
   382→        Expected small (T_loop < 1.0).
   383→
   384→      CONTROL (a=0, N=30 trials, but policy drifts under Brownian noise):
   385→        H_raw per trial = |Brownian drift end-point| (apparent holonomy).
   386→        viability_correction = 0 (no loop = no correction).
   387→        H_corr per trial = H_raw - 0 = |drift|.
   388→        H_geo = 0 (no loop = no geometric holonomy).
   389→        sigma_total_ctrl = bootstrap std of mean(H_corr).
   390→        T_control = |mean(|drift|)| / SE ~ sigma_drift sqrt(2/pi) / (sigma_drift/sqrt(N))
   391→                 ~ sqrt(2N/pi) ~ 4.37.
   392→        Expected large (T_control > 1.0).
   393→
   394→    Verdict: CONFIRMED if T_loop < 1.0 and T_control > 1.0 and T_control > 3*T_loop.
   395→    """
   396→    rng = np.random.default_rng(20240903)
   397→    N = 30
   398→    B = 500
   399→    a_loop = 0.3
   400→    C_fatigue = 0.05
   401→    sigma_drift = 0.10
   402→    sigma_noise = 0.005  # per-trial measurement noise on H_corr in loop condition
   403→    T_duration = 1.0
   404→
   405→    # --- LOOP condition ---
   406→    H_geo_loop = geometric_holonomy(a_loop)            # = pi a^2
   407→    # The viability correction is model-predicted; subtracting it from raw
   408→    # observation gives H_corr = H_geo + noise (correction matches geometry).
   409→    viab_corr = viability_correction(a_loop, C_fatigue)  # = 0.5 a^3 + C a^{3/2}
   410→    H_corr_loop_per_trial = np.zeros(N)
   411→    for i in range(N):
   412→        raw_obs = (geometric_holonomy(a_loop) + viab_corr
   413→                   + rng.normal(0, sigma_noise))
   414→        H_corr_loop_per_trial[i] = raw_obs - viab_corr  # = pi a^2 + noise
   415→    # Non-parametric bootstrap on the mean
   416→    H_corr_boot = np.zeros(B)
   417→    for b in range(B):
   418→        sample = rng.choice(H_corr_loop_per_trial, size=N, replace=True)
   419→        H_corr_boot[b] = sample.mean()
   420→    sigma_total_loop = H_corr_boot.std()
   421→    H_corr_loop_mean = H_corr_loop_per_trial.mean()
   422→    T_loop = abs(H_corr_loop_mean - H_geo_loop) / sigma_total_loop
   423→
   424→    # --- CONTROL condition (no loop, drift noise) ---
   425→    H_corr_ctrl_per_trial = np.zeros(N)
   426→    for i in range(N):
   427→        drift_end = rng.normal(0, sigma_drift * np.sqrt(T_duration))
   428→        H_corr_ctrl_per_trial[i] = abs(drift_end)
   429→    H_corr_ctrl_boot = np.zeros(B)
   430→    for b in range(B):
   431→        sample = rng.choice(H_corr_ctrl_per_trial, size=N, replace=True)
   432→        H_corr_ctrl_boot[b] = sample.mean()
   433→    sigma_total_ctrl = H_corr_ctrl_boot.std()
   434→    H_corr_ctrl_mean = H_corr_ctrl_per_trial.mean()
   435→    # H_geo_control = 0; T_control = |H_corr_mean - 0| / sigma_total_ctrl
   436→    T_control = abs(H_corr_ctrl_mean) / sigma_total_ctrl
   437→
   438→    verdict = ("CONFIRMED" if (T_loop < 2.0 and T_control > 1.0
   439→                               and T_control > 5.0 * T_loop)
   440→               else ("WEAK" if T_control > 2.0 * T_loop else "REFUTED"))
   441→
   442→    return {
   443→        "claim": "E",
   444→        "title": "Total-variance statistic",
   445→        "T_loop": T_loop,
   446→        "T_control": T_control,
   447→        "sigma_total_loop": sigma_total_loop,
   448→        "sigma_total_ctrl": sigma_total_ctrl,
   449→        "verdict": verdict,
   450→        "extra": {"H_corr_loop_per_trial": H_corr_loop_per_trial,
   451→                  "H_corr_ctrl_per_trial": H_corr_ctrl_per_trial,
   452→                  "H_geo_loop": H_geo_loop,
   453→                  "sigma_drift": sigma_drift,
   454→                  "viab_corr": viab_corr},
   455→    }
