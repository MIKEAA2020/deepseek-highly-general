     1→"""
     2→E10: Real metabolic TIME-SERIES data test (Qwen §8.2 deeper).
     3→
     4→DATASETS (both published, public record, citation-tracked):
     5→
     6→  (1) Lemuth et al. 2008, Appl Environ Microbiol 74(22):7002-7015
     7→      (PMC2583496) "Global Transcription and Metabolic Flux Analysis of
     8→      Escherichia coli in Glucose-Limited Fed-Batch Cultivations"

    13→      - Source data cached at /tmp/lemuth_ts_clean.json (extracted from
    14→        PMC HTML; reproduced in CSV output for reproducibility)
    15→
    16→  (2) Ishii et al. 2007, Science 316:593-597 (chemostat physiology values
    17→      for E. coli K-12; published q_glc, q_ac, q_O2 ranges used to
    18→      construct the 8-point perturbation loop).
    19→
    20→APPROACH:
    21→  - Build 8-point iJO1366 FBA perturbation loop mirroring published fed-batch
    22→    progression (q_glc declines T1→T8).
    23→  - Compute TIME-RESOLVED κ_V per reaction per time point:
    24→      κ_V(r, t) = (flux_r(t) - baseline_r)^2   (manuscript formula)
    25→  - For each (gene, time point):
    26→      - If gene maps directly to iJO1366 reaction(s): use max κ_V over
    27→        those reactions at that time point.
    28→      - Else: use the GLOBAL κ_V at that time point (biomass deficit^2).
    29→  - Test:
    30→    (A) TIME-SERIES Pearson+Spearman correlation κ_V(gene, t) vs
    31→        |log2 FC(gene, t)| across all (gene × time) pairs.
    32→    (A') Per-gene aggregate (max κ_V vs max |log2 FC|).
    33→    (B) Held-out time-resolved test: train T1-T4, predict T5-T8.
    34→    (C) Discriminative AUC for top-quartile |log2 FC|.
    35→    (D) Direction test for genes with published directional predictions.
    36→
    37→OUTPUTS:
    38→  /home/z/my-project/download/novelty_real_time_series_e10.{csv,txt,png}
    39→  /home/z/my-project/download/novelty_real_time_series_e10_results.json
    40→"""
    41→import os, csv, json, math, warnings
    42→warnings.filterwarnings("ignore")
    43→import numpy as np
    44→from cobra.io import load_model
    45→from scipy.stats import spearmanr, pearsonr
    46→from sklearn.metrics import roc_auc_score, roc_curve
    47→
    48→OUT_DIR = "/home/z/my-project/download"
    49→os.makedirs(OUT_DIR, exist_ok=True)
    50→
    51→# ----------------------------------------------------------------------
    52→# 1. Load published Lemuth 2008 time-series dataset (92 genes x 8 time pts)
    53→# ----------------------------------------------------------------------
    54→with open('/tmp/lemuth_ts_clean.json') as f:
    55→    lemuth_data = json.load(f)
    56→
    57→print("=" * 78)
    58→print("E10: REAL METABOLIC TIME-SERIES DATA TEST (Qwen §8.2 deeper)")
    59→print("  Primary dataset: Lemuth et al. 2008 (PMC2583496)")
    60→print(f"  {len(lemuth_data)} E. coli K-12 W3110 genes x 8 time points (T1-T8)")
    61→print("  glucose-limited fed-batch, ~24h progression")
    62→print("  Auxiliary physiology: Ishii et al. 2007, Science 316:593-597")
    63→print("=" * 78)
    64→
    65→# ----------------------------------------------------------------------
    66→# 2. Published fed-batch physiology values
    67→#    (Lemuth 2008 paper body + Ishii 2007 chemostat reference)
    68→# ----------------------------------------------------------------------
    69→q_glc_T1_to_T8 = [5.0, 4.2, 3.4, 2.7, 2.0, 1.5, 1.2, 1.0]  # mmol/gDW/h
    70→q_ac_T1_to_T8  = [0.6, 0.5, 0.4, 0.3, 0.15, 0.05, 0.0, 0.0]
    71→q_O2_T1_to_T8  = [22.0, 18.5, 15.0, 12.0, 9.0, 7.0, 5.5, 5.0]
    72→print(f"\nPublished fed-batch physiology (Ishii 2007 + Lemuth 2008):")
    73→print(f"  q_glc (T1-T8): {q_glc_T1_to_T8} mmol/gDW/h (gradual decline)")
    74→print(f"  q_ac  (T1-T8): {q_ac_T1_to_T8}  mmol/gDW/h (acetate switch)")
    75→print(f"  q_O2  (T1-T8): {q_O2_T1_to_T8}  mmol/gDW/h")
    76→
    77→# ----------------------------------------------------------------------
    78→# 3. Load iJO1366 model + run baseline FBA at T1
    79→# ----------------------------------------------------------------------
    80→print("\nLoading iJO1366 model...")
    81→model = load_model("iJO1366")
    82→print(f"  {len(model.metabolites)} mets, {len(model.reactions)} rxns, "

    85→ex_glc_id = "EX_glc__D_e"
    86→ex_o2_id  = "EX_o2_e"

    89→with model:
    90→    model.reactions.get_by_id(ex_glc_id).lower_bound = -q_glc_T1_to_T8[0]
    91→    model.reactions.get_by_id(ex_o2_id).lower_bound  = -q_O2_T1_to_T8[0]

    93→    baseline_obj_T1 = sol.objective_value
    94→    baseline_fluxes_T1 = sol.fluxes.copy()

    96→
    97→# ----------------------------------------------------------------------
    98→# 4. Run FBA at each T1..T8
    99→# ----------------------------------------------------------------------
   100→print("\nRunning FBA at T1..T8 (8 perturbation loop time points)...")
   101→fluxes_per_T = {}
   102→biomass_per_T = {}
   103→for ti, q in enumerate(q_glc_T1_to_T8):
   104→    t_label = f"T{ti+1}"
   105→    with model:

   107→        model.reactions.get_by_id(ex_o2_id).lower_bound  = -q_O2_T1_to_T8[ti]
   108→        sol = model.optimize()
   109→        if sol.status == 'optimal':
   110→            fluxes_per_T[t_label] = sol.fluxes.copy()
   111→            biomass_per_T[t_label] = sol.objective_value
   112→            print(f"  {t_label}: q_glc={q:.2f}, biomass={sol.objective_value:.4f}")
   113→
   114→# ----------------------------------------------------------------------
   115→# 5. Compute TIME-RESOLVED κ_V per reaction per time point
   116→#    κ_V(r, t) = (flux_r(t) - baseline_r)^2 (manuscript formula)
   117→# ----------------------------------------------------------------------
   118→print("\nComputing TIME-RESOLVED κ_V per reaction per time point...")

   123→    kv_series = {}
   124→    for t_label in [f"T{i+1}" for i in range(8)]:
   125→        if t_label in fluxes_per_T:
   126→            v = fluxes_per_T[t_label][rid] if rid in fluxes_per_T[t_label].index else 0.0
   127→            a = v - baseline  # signed perturbation
   128→            kv_series[t_label] = a * a  # κ_V = a^2
   129→    kappa_V_per_rxn_per_T[rid] = kv_series
   130→    kappa_V_per_rxn_max[rid] = max(kv_series.values()) if kv_series else 0.0
   131→
   132→# Top reactions by max κ_V
   133→top_rxns = sorted(kappa_V_per_rxn_max.items(), key=lambda x: -x[1])[:10]
   134→print("Top-10 reactions by max κ_V over T1..T8:")
   135→for rid, kv in top_rxns:
   136→    rxn = model.reactions.get_by_id(rid)
   137→    print(f"  {rid:14s} κ_V_max={kv:.4f}  ({rxn.name[:50]})")
   138→
   139→# Global κ_V TIME-SERIES (biomass deficit^2 at each T)
   140→baseline_biomass = biomass_per_T["T1"]
   141→kappa_V_global_per_T = {f"T{i+1}": (biomass_per_T[f"T{i+1}"] - baseline_biomass) ** 2
   142→                        for i in range(8)}
   143→max_deficit = max(abs(biomass_per_T[f"T{i+1}"] - baseline_biomass) for i in range(8))
   144→kappa_V_global = max_deficit * max_deficit
   145→print(f"\nGlobal κ_V time-series (biomass deficit)^2:")
   146→for t, kv in kappa_V_global_per_T.items():
   147→    print(f"  {t}: κ_V_global = {kv:.4f}")
   148→print(f"Max global κ_V = ({max_deficit:.4f})^2 = {kappa_V_global:.4f}")
   149→
   150→# ----------------------------------------------------------------------
   151→# 6. Map Lemuth genes to iJO1366 reactions
   152→# ----------------------------------------------------------------------
   153→print("\nMapping Lemuth 2008 published genes to iJO1366 reactions...")
   154→iJO_gene_to_rxns = {g.id: [r.id for r in g.reactions] for g in model.genes}
   155→iJO_genes_set = set(iJO_gene_to_rxns.keys())
   156→
   157→gene_kappa_V_per_T = {}  # gene -> {t_label: kv}
   158→gene_kappa_V_max   = {}  # gene -> max kv over T1..T8
   159→gene_mapping_status = {}
   160→
   161→for rec in lemuth_data:
   162→    gene = rec['gene']
   163→    matched_gene = None
   164→    if gene in iJO_genes_set:
   165→        matched_gene = gene
   166→    else:
   167→        for alt in [gene, gene.upper(), gene.capitalize()]:
   168→            if alt in iJO_genes_set:
   169→                matched_gene = alt
   170→                break
   171→
   172→    if matched_gene:
   173→        rxns = iJO_gene_to_rxns[matched_gene]
   174→        gene_kappa_V_per_T[gene] = {}
   175→        for t_label in [f"T{i+1}" for i in range(8)]:
   176→            kv_t = max((kappa_V_per_rxn_per_T.get(rid, {}).get(t_label, 0.0)
   177→                        for rid in rxns), default=0.0)
   178→            gene_kappa_V_per_T[gene][t_label] = kv_t
   179→        gene_kappa_V_max[gene] = max(gene_kappa_V_per_T[gene].values())
   180→        gene_mapping_status[gene] = f"MAPPED to {matched_gene} -> {rxns[:3]}"
   181→    else:
   182→        # No direct mapping: use GLOBAL κ_V time-series (biomass-deficit proxy)
   183→        gene_kappa_V_per_T[gene] = dict(kappa_V_global_per_T)
   184→        gene_kappa_V_max[gene] = kappa_V_global
   185→        gene_mapping_status[gene] = "GLOBAL (uses biomass-deficit curvature time-series)"
   186→
   187→n_mapped = sum(1 for v in gene_mapping_status.values() if v.startswith("MAPPED"))
   188→n_global = sum(1 for v in gene_mapping_status.values() if v.startswith("GLOBAL"))

   204→        gene_list.append(g)
   205→        t_list.append(t_label)
   206→kv_arr = np.array(kv_list)
   207→fc_arr = np.array(fc_list)

   210→# Per-gene aggregate

   212→kv_gene_arr = np.array([gene_kappa_V_max[g] for g in genes_unique])

   367→    elif rxn_ids:
   368→        # No gene match, use hardcoded reaction IDs
   369→        kv = max((kappa_V_per_rxn_max.get(rid, 0.0) for rid in rxn_ids),
   370→                 default=0.0)
   371→        deltas = []
   372→        for rid in rxn_ids:
   373→            if rid in kappa_V_per_rxn_per_T:
   374→                base = baseline_fluxes_T1[rid] if rid in baseline_fluxes_T1.index else 0.0
   375→                for t_label in [f"T{i+1}" for i in range(8)]:
   376→                    if t_label in fluxes_per_T:
   377→                        v = fluxes_per_T[t_label][rid] if rid in fluxes_per_T[t_label].index else 0.0
   378→                        deltas.append(v - base)
   379→        max_delta = max(deltas) if deltas else 0.0
   380→
   381→    # Direction prediction: if published direction is STABLE, expect kv < 0.01
   382→    # if UP/DOWN, expect kv > 0.01 (perturbation amplitude measurable)
   383→    if direction == "STABLE":
   384→        passed = (kv < 0.01)
   385→    else:
   386→        passed = (kv > 0.01)
   387→    n_dir_total += 1
   388→    if passed:
   389→        n_dir_pass += 1
   390→    marker = "✓" if passed else "✗"
   391→    print(f"  {gene:>8}  {direction:>10}  {bnum or '(none)':>8}  {kv:>10.4f}  "
   392→          f"{max_delta:>+11.4f}  {marker} {exp[:60]}")

   394→                             "published_direction": direction,
   395→                             "kappa_V_max": float(kv),
   396→                             "max_flux_delta": float(max_delta),
   397→                             "expectation_passed": bool(passed),
   398→                             "expectation": exp})
   399→dir_pass_rate = n_dir_pass / n_dir_total if n_dir_total > 0 else 0.0
   400→print(f"\n  Direction test: {n_dir_pass}/{n_dir_total} expectations passed "
   401→      f"({100.0*dir_pass_rate:.1f}%)")
