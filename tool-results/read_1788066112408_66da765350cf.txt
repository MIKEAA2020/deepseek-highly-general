   346→    knockout_reactions = set()
   347→    if knockout_species is not None:
   348→        for r in rxs:
   349→            produced = [s for s, d in r["stoich"].items() if d > 0]
   350→            if knockout_species in produced:
   351→                knockout_reactions.add(r["id"])
   352→
   353→    dt = 0.05
   354→    trajectory = [{s: x[s] for s in sp}]
   355→    for step in range(T):
   356→        rates = {}
   357→        for r in rxs:
   358→            if r["id"] in knockout_reactions:
   359→                rates[r["id"]] = 0.0
   360→                continue
   361→            subs = [(s, abs(d)) for s, d in r["stoich"].items() if d < 0]
   362→            if any(x[s] <= 0 for s, _ in subs):
   363→                rates[r["id"]] = 0.0
   364→                continue
   365→            kind = r["kind"]
   366→            cat = r["catalyst"]
   367→            if kind == "constitutive":
   368→                k = k_cat_constitutive
   369→                rate = k
   370→                for s, _ in subs:
   371→                    rate *= x[s] / (Km + x[s])
   372→            else:
   373→                cat_conc = x.get(cat, 0.0) if cat is not None else 0.0
   374→                if cat_conc <= 0:
   375→                    rates[r["id"]] = 0.0
   376→                    continue
   377→                if kind == "synthesis":
   378→                    k = k_cat_synthesis
   379→                elif kind == "autocatalytic":
   380→                    k = k_cat_autocatalytic
   381→                else:
   382→                    k = k_cat_metabolic
   383→                # Per-reaction k_cat override (e.g., for boosted ALT/ASPAT/M11)
   384→                if "k_cat_override" in r:
   385→                    k = r["k_cat_override"]
   386→                rate = k * cat_conc
   387→                for s, _ in subs:
   388→                    rate *= x[s] / (Km + x[s])
   389→            rates[r["id"]] = rate
   390→
   391→        dx = {s: 0.0 for s in sp}
   392→        for r in rxs:
   393→            rate = rates[r["id"]]
   394→            for s, d in r["stoich"].items():
   395→                dx[s] += d * rate
   396→        for s in sp:
   397→            dx[s] -= delta * x[s]
   398→            if s in fd:
   399→                dx[s] += food_supply_rate * (food_conc - x[s]) * 0.5
   400→        for s in sp:
   401→            x[s] = max(0.0, min(max_conc, x[s] + dt * dx[s]))
   402→        trajectory.append({s: x[s] for s in sp})
   403→    return trajectory

   406→def simulate_network_recover(network, init, T=500, delta=0.05,
   407→                             k_cat_metabolic=0.8, k_cat_synthesis=2.0,
   408→                             k_cat_constitutive=0.3, k_cat_autocatalytic=1.0,
   409→                             food_conc=10.0, food_supply_rate=2.0,
   410→                             Km=0.1, max_conc=100.0):
   411→    """Simulate from a given initial condition (recovery test)."""
   412→    sp = network["species"]
   413→    fd = set(network["food"])
   414→    rxs = network["reactions"]
   415→    x = {s: init.get(s, 0.0) for s in sp}
   416→    for s in fd:
   417→        x[s] = max(x[s], food_conc)
   418→    if x.get("ATP", 0) < 1.0:
   419→        x["ATP"] = max(x["ATP"], 2.0)
   420→    if x.get("TF", 0) < 0.05:
   421→        x["TF"] = max(x["TF"], 0.1)
   422→    dt = 0.05
   423→    trajectory = [{s: x[s] for s in sp}]
   424→    for step in range(T):
   425→        rates = {}
   426→        for r in rxs:
   427→            subs = [(s, abs(d)) for s, d in r["stoich"].items() if d < 0]
   428→            if any(x[s] <= 0 for s, _ in subs):
   429→                rates[r["id"]] = 0.0
   430→                continue
   431→            kind = r["kind"]
   432→            cat = r["catalyst"]
   433→            if kind == "constitutive":
   434→                k = k_cat_constitutive
   435→                rate = k
   436→                for s, _ in subs:
   437→                    rate *= x[s] / (Km + x[s])
   438→            else:
   439→                cat_conc = x.get(cat, 0.0) if cat is not None else 0.0
   440→                if cat_conc <= 0:
   441→                    rates[r["id"]] = 0.0
   442→                    continue
   443→                if kind == "synthesis":
   444→                    k = k_cat_synthesis
   445→                elif kind == "autocatalytic":
   446→                    k = k_cat_autocatalytic
   447→                else:
   448→                    k = k_cat_metabolic
   449→                # Per-reaction k_cat override (e.g., for boosted ALT/ASPAT/M11)
   450→                if "k_cat_override" in r:
   451→                    k = r["k_cat_override"]
   452→                rate = k * cat_conc
   453→                for s, _ in subs:
   454→                    rate *= x[s] / (Km + x[s])
   455→            rates[r["id"]] = rate
   456→        dx = {s: 0.0 for s in sp}
   457→        for r in rxs:
   458→            rate = rates[r["id"]]
   459→            for s, d in r["stoich"].items():
   460→                dx[s] += d * rate
   461→        for s in sp:
   462→            dx[s] -= delta * x[s]
   463→            if s in fd:
   464→                dx[s] += food_supply_rate * (food_conc - x[s]) * 0.5
   465→        for s in sp:
   466→            x[s] = max(0.0, min(max_conc, x[s] + dt * dx[s]))
   467→        trajectory.append({s: x[s] for s in sp})
   468→    return trajectory

   471→def closure_test(network, network_name, T=500, viability_threshold=0.1):
   472→    """Run the autopoiesis closure test of Definition def:autopoiesis."""
   473→    baseline = simulate_network(network, knockout_species=None, T=T)
   474→    baseline_final = baseline[-1]
   475→    records = []
   476→    for m_j in network["non_food"]:
   477→        knock = simulate_network(network, knockout_species=m_j, T=T)
   478→        knock_final = knock[-1]
   479→        knock_traj = [t[m_j] for t in knock]
   480→        recover_start_idx = T // 2
   481→        recover_init = knock[recover_start_idx]
   482→        recover = simulate_network_recover(network, init=recover_init, T=T - recover_start_idx)
   483→        recover_final = recover[-1]
   484→        knock_success = knock_final[m_j] < viability_threshold
   485→        recover_success = recover_final[m_j] > viability_threshold
   486→        causally_internal = knock_success and recover_success
   487→        records.append({
   488→            "network": network_name,
   489→            "component": m_j,
   490→            "baseline_conc": baseline_final[m_j],
   491→            "knockout_conc_final": knock_final[m_j],
   492→            "knockout_min": min(knock_traj),
   493→            "recover_conc_final": recover_final[m_j],
   494→            "knockout_success": knock_success,
   495→            "recover_success": recover_success,
   496→            "causally_internal": causally_internal,
   497→            "verdict": "AUTOPOIETIC" if causally_internal else "HOMEOSTATIC",
   498→        })
   499→    return records, baseline
