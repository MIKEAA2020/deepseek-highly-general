   318→def simulate_network(network, knockout_species=None, T=500, delta=0.05,
   319→                     k_cat_metabolic=0.8, k_cat_synthesis=2.0,
   320→                     k_cat_constitutive=0.3, k_cat_autocatalytic=1.0,
   321→                     food_supply_rate=2.0, food_conc=10.0,
   322→                     Km=0.1, max_conc=100.0,
   323→                     init_concs=None, allow_constitutive=True):
   324→    """Simulate the network dynamics for T steps.
   325→
   326→    Uses Michaelis-Menten saturation for each substrate to prevent
   327→    runaway positive feedback: rate = k * [cat] * prod(s_i / (Km + s_i)).
   328→    This caps the rate at k * [cat] when all substrates are saturating,
   329→    avoiding the unbounded growth that arises from mass-action kinetics
   330→    in a network with multiple positive-feedback loops.
   331→
   332→    Concentrations are also capped at max_conc to prevent numerical
   333→    overflow during transient excursions.
   334→    """
   335→    sp = network["species"]
   336→    fd = set(network["food"])
   337→    rxs = network["reactions"]
   338→
   339→    # Initial concentrations: food at food_conc, others at small seed value
   340→    x = {s: (food_conc if s in fd else 0.1) for s in sp}
   341→    if init_concs:
   342→        for s, v in init_concs.items():
   343→            x[s] = v
   344→
   345→    knockout_reactions = set()
   346→    if knockout_species is not None:
   347→        for r in rxs:
   348→            produced = [s for s, d in r["stoich"].items() if d > 0]
   349→            if knockout_species in produced:
   350→                knockout_reactions.add(r["id"])
   351→
   352→    dt = 0.05
   353→    trajectory = [{s: x[s] for s in sp}]
   354→    for step in range(T):
   355→        rates = {}
   356→        for r in rxs:
   357→            if r["id"] in knockout_reactions:
   358→                rates[r["id"]] = 0.0
   359→                continue
   360→            subs = [(s, abs(d)) for s, d in r["stoich"].items() if d < 0]
   361→            if any(x[s] <= 0 for s, _ in subs):
   362→                rates[r["id"]] = 0.0
   363→                continue
   364→            kind = r["kind"]
   365→            cat = r["catalyst"]
   366→            if kind == "constitutive":
   367→                # No catalyst: rate = k_const * prod(s_i / (Km + s_i))
   368→                k = k_cat_constitutive
   369→                rate = k
   370→                for s, _ in subs:
   371→                    rate *= x[s] / (Km + x[s])
   372→            else:
   373→                # Catalyzed: rate = k * [cat] * prod(s_i / (Km + s_i))
   374→                cat_conc = x.get(cat, 0.0) if cat is not None else 0.0
   375→                if cat_conc <= 0:
   376→                    rates[r["id"]] = 0.0
   377→                    continue
   378→                if kind == "synthesis":
   379→                    k = k_cat_synthesis
   380→                elif kind == "autocatalytic":
   381→                    k = k_cat_autocatalytic
   382→                else:  # metabolic
   383→                    k = k_cat_metabolic
   384→                rate = k * cat_conc
   385→                for s, _ in subs:
   386→                    rate *= x[s] / (Km + x[s])
   387→            rates[r["id"]] = rate
   388→
   389→        dx = {s: 0.0 for s in sp}
   390→        for r in rxs:
   391→            rate = rates[r["id"]]
   392→            for s, d in r["stoich"].items():
   393→                dx[s] += d * rate
   394→        # First-order degradation
   395→        for s in sp:
   396→            dx[s] -= delta * x[s]
   397→            # Food supply
   398→            if s in fd:
   399→                dx[s] += food_supply_rate * (food_conc - x[s]) * 0.5
   400→        # Update with positivity and max-conc cap
   401→        for s in sp:
   402→            x[s] = max(0.0, min(max_conc, x[s] + dt * dx[s]))
   403→        trajectory.append({s: x[s] for s in sp})
   404→    return trajectory

   407→def simulate_network_recover(network, init, T=500, delta=0.05,
   408→                             k_cat_metabolic=0.8, k_cat_synthesis=2.0,
   409→                             k_cat_constitutive=0.3, k_cat_autocatalytic=1.0,
   410→                             food_conc=10.0, food_supply_rate=2.0,
   411→                             Km=0.1, max_conc=100.0):
   412→    """Simulate from a given initial condition (recovery test)."""
   413→    sp = network["species"]
   414→    fd = set(network["food"])
   415→    rxs = network["reactions"]
   416→    x = {s: init.get(s, 0.0) for s in sp}
   417→    # Ensure food is at supply level at start of recovery
   418→    for s in fd:
   419→        x[s] = max(x[s], food_conc)
   420→    # Seed a small amount of ATP and TF to bootstrap recovery from depleted states
   421→    # (represents stored energy reserves + basal transcription machinery)
   422→    if x.get("ATP", 0) < 1.0:
   423→        x["ATP"] = max(x["ATP"], 2.0)
   424→    if x.get("TF", 0) < 0.05:
   425→        x["TF"] = max(x["TF"], 0.1)
   426→    dt = 0.05
   427→    trajectory = [{s: x[s] for s in sp}]
   428→    for step in range(T):
   429→        rates = {}
   430→        for r in rxs:
   431→            subs = [(s, abs(d)) for s, d in r["stoich"].items() if d < 0]
   432→            if any(x[s] <= 0 for s, _ in subs):
   433→                rates[r["id"]] = 0.0
   434→                continue
   435→            kind = r["kind"]
   436→            cat = r["catalyst"]
   437→            if kind == "constitutive":
   438→                k = k_cat_constitutive
   439→                rate = k
   440→                for s, _ in subs:
   441→                    rate *= x[s] / (Km + x[s])
   442→            else:
   443→                cat_conc = x.get(cat, 0.0) if cat is not None else 0.0
   444→                if cat_conc <= 0:
   445→                    rates[r["id"]] = 0.0
   446→                    continue
   447→                if kind == "synthesis":
   448→                    k = k_cat_synthesis
   449→                elif kind == "autocatalytic":
   450→                    k = k_cat_autocatalytic
   451→                else:
   452→                    k = k_cat_metabolic
   453→                rate = k * cat_conc
   454→                for s, _ in subs:
   455→                    rate *= x[s] / (Km + x[s])
   456→            rates[r["id"]] = rate
   457→        dx = {s: 0.0 for s in sp}
   458→        for r in rxs:
   459→            rate = rates[r["id"]]
   460→            for s, d in r["stoich"].items():
   461→                dx[s] += d * rate
   462→        for s in sp:
   463→            dx[s] -= delta * x[s]
   464→            if s in fd:
   465→                dx[s] += food_supply_rate * (food_conc - x[s]) * 0.5
   466→        for s in sp:
   467→            x[s] = max(0.0, min(max_conc, x[s] + dt * dx[s]))
   468→        trajectory.append({s: x[s] for s in sp})
   469→    return trajectory

   472→def closure_test(network, network_name, T=500, viability_threshold=0.1):
   473→    """Run the autopoiesis closure test of Definition def:autopoiesis."""
   474→    baseline = simulate_network(network, knockout_species=None, T=T)
   475→    baseline_final = baseline[-1]
   476→    records = []
   477→    for m_j in network["non_food"]:
   478→        knock = simulate_network(network, knockout_species=m_j, T=T)
   479→        knock_final = knock[-1]
   480→        knock_traj = [t[m_j] for t in knock]
   481→        # Recovery test
   482→        recover_start_idx = T // 2
   483→        recover_init = knock[recover_start_idx]
   484→        recover = simulate_network_recover(network, init=recover_init, T=T - recover_start_idx)
   485→        recover_final = recover[-1]
   486→        knock_success = knock_final[m_j] < viability_threshold
   487→        recover_success = recover_final[m_j] > viability_threshold
   488→        causally_internal = knock_success and recover_success
   489→        records.append({
   490→            "network": network_name,
   491→            "component": m_j,
   492→            "baseline_conc": baseline_final[m_j],
   493→            "knockout_conc_final": knock_final[m_j],
   494→            "knockout_min": min(knock_traj),
   495→            "recover_conc_final": recover_final[m_j],
   496→            "knockout_success": knock_success,
   497→            "recover_success": recover_success,
   498→            "causally_internal": causally_internal,
   499→            "verdict": "AUTOPOIETIC" if causally_internal else "HOMEOSTATIC",
   500→        })
   501→    return records, baseline
