    91→# ----------------------------------------------------------------------
    92→# so(3) generators (3x3 antisymmetric real matrices)
    93→#   [T_x, T_y] = T_z,  [T_y, T_z] = T_x,  [T_z, T_x] = T_y
    94→# ----------------------------------------------------------------------
    95→T_x = np.array([[0.0,  0.0, 0.0],
    96→                [0.0,  0.0, -1.0],
    97→                [0.0,  1.0, 0.0]])
    98→T_y = np.array([[0.0,  0.0, 1.0],
    99→                [0.0,  0.0, 0.0],
   100→                [-1.0, 0.0, 0.0]])
   101→T_z = np.array([[0.0, -1.0, 0.0],
   102→                [1.0,  0.0, 0.0],
   103→                [0.0,  0.0, 0.0]])
   104→
   105→# Sanity: commutators
   106→assert np.allclose(T_x @ T_y - T_y @ T_x, T_z), "[Tx, Ty] != Tz"
   107→assert np.allclose(T_y @ T_z - T_z @ T_y, T_x), "[Ty, Tz] != Tx"
   108→assert np.allclose(T_z @ T_x - T_x @ T_z, T_y), "[Tz, Tx] != Ty"

   110→I3 = np.eye(3)
   111→
   112→# ----------------------------------------------------------------------
   113→# Setup: two-stratum base, connection on strata, boundary transition
   114→# ----------------------------------------------------------------------
   115→F_curvature = 2.0  # constant curvature magnitude on both strata (F_+ = F_- = F)
   116→eps_values = [0.05, 0.1, 0.2, 0.4, 0.8]  # loop side length sweep
   117→y_c = 0.0  # loop center y EXACTLY ON boundary, so loop straddles x-axis
   118→x_c = 0.5  # loop center x-coordinate
   119→a_1 = 1.5  # slope of alpha(x) = a_1 * x
   120→
   121→def alpha(x):
   122→    """Boundary transition function alpha(x); rotation angle about y-axis."""
   123→    return a_1 * x
   124→
   125→
   126→def A_conn(x, y, dx, dy):
   127→    """so(3)-valued connection 1-form evaluated at (x,y) on a segment (dx,dy).
   128→    Returns the 3x3 matrix integral of A = (F/2)(x dy - y dx) T_z along the
   129→    straight segment with midpoint (x,y) and displacement (dx,dy). For the
   130→    midpoint-rule approximation we evaluate at the segment midpoint."""
   131→    return (F_curvature / 2.0) * (x * dy - y * dx) * T_z
   132→
   133→
   134→def g_boundary(x, alpha_fn=None):
   135→    """SO(3) boundary transition matrix g_{+-}(x,0) = exp(alpha(x) T_y)."""
   136→    a = alpha_fn if alpha_fn is not None else alpha
   137→    return expm(a(x) * T_y)
   138→
   139→
   140→def g_boundary_inv(x, alpha_fn=None):
   141→    """g_{+-}(x,0)^{-1} = exp(-alpha(x) T_y)."""
   142→    a = alpha_fn if alpha_fn is not None else alpha
   143→    return expm(-a(x) * T_y)

   146→# ----------------------------------------------------------------------
   147→# Analytic piecewise holonomy (closed form for the rectangular loop)
   148→# ----------------------------------------------------------------------
   149→def stratified_holonomy_analytic_so3(x_c, y_c, eps, F, alpha_fn):
   150→    """Closed-form SO(3) piecewise holonomy for the rectangular loop.
   151→
   152→    Loop corners (counterclockwise from bottom-left):
   153→      BL = (x_c - eps/2, -eps/2)
   154→      BR = (x_c + eps/2, -eps/2)
   155→      TR = (x_c + eps/2, +eps/2)
   156→      TL = (x_c - eps/2, +eps/2)
   157→
   158→    Traversal: BL -> BR -> TR -> TL -> BL, crossing the x-axis at
   159→      p_+ = (x_c + eps/2, 0)   (S_- -> S_+ on the right edge)
   160→      p_- = (x_c - eps/2, 0)   (S_+ -> S_- on the left edge)
   161→
   162→    The piecewise holonomy is (left-to-right = last-to-first applied;
   163→    the first segment's matrix is the rightmost):
   164→      H = Hol(p_- -> BL) * g(p_-) * Hol(TL -> p_-) * Hol(TR -> TL) *
   165→          Hol(p_+ -> TR) * g(p_+)^{-1} * Hol(BR -> p_+) * Hol(BL -> BR)
   166→
   167→    where Hol(segment) = expm(-(F/2) * integral of (x dy - y dx) along segment * T_z)
   168→    (Schrödinger convention: psi -> expm(-A_int * T_z) psi).
   169→    """
   170→    p_plus_x = x_c + eps / 2  # x-coord of p_+
   171→    p_minus_x = x_c - eps / 2  # x-coord of p_-
   172→
   173→    # Edge 1: BL -> BR (S_-), y = -eps/2, dy = 0, dx = eps, midpoint (x_c, -eps/2)
   174→    # integral = (F/2)*(x_c*0 - (-eps/2)*eps) = F*eps^2/4
   175→    H_1 = expm(-F * eps**2 / 4 * T_z)
   176→    # Edge 2 part 1: BR -> p_+ (S_-), x = p_plus_x, dy = eps/2 (from -eps/2 to 0),
   177→    #   dx = 0, midpoint (p_plus_x, -eps/4)
   178→    # integral = (F/2)*(p_plus_x * (eps/2) - (-eps/4)*0) = F*eps*p_plus_x/4
   179→    H_21 = expm(-F * eps * p_plus_x / 4 * T_z)
   180→    # Boundary crossing p_+: S_- -> S_+ => apply g(p_+)^{-1} = exp(-alpha(p_+) T_y)
   181→    H_cross_p_plus = g_boundary_inv(p_plus_x, alpha_fn)
   182→    # Edge 2 part 2: p_+ -> TR (S_+), x = p_plus_x, dy = eps/2, dx = 0,
   183→    #   midpoint (p_plus_x, eps/4)
   184→    # integral = F*eps*p_plus_x/4
   185→    H_22 = expm(-F * eps * p_plus_x / 4 * T_z)
   186→    # Edge 3: TR -> TL (S_+), y = eps/2, dy = 0, dx = -eps, midpoint (x_c, eps/2)
   187→    # integral = (F/2)*(x_c*0 - (eps/2)*(-eps)) = F*eps^2/4
   188→    H_3 = expm(-F * eps**2 / 4 * T_z)
   189→    # Edge 4 part 1: TL -> p_- (S_+), x = p_minus_x, dy = -eps/2 (from eps/2 to 0),
   190→    #   dx = 0, midpoint (p_minus_x, eps/4)
   191→    # integral = (F/2)*(p_minus_x*(-eps/2) - (eps/4)*0) = -F*eps*p_minus_x/4
   192→    H_41 = expm(-(-F * eps * p_minus_x / 4) * T_z)  # = expm(+F*eps*p_minus_x/4 * T_z)
   193→    # Boundary crossing p_-: S_+ -> S_- => apply g(p_-) = exp(+alpha(p_-) T_y)
   194→    H_cross_p_minus = g_boundary(p_minus_x, alpha_fn)
   195→    # Edge 4 part 2: p_- -> BL (S_-), x = p_minus_x, dy = -eps/2 (from 0 to -eps/2),
   196→    #   dx = 0, midpoint (p_minus_x, -eps/4)
   197→    # integral = (F/2)*(p_minus_x*(-eps/2) - (-eps/4)*0) = -F*eps*p_minus_x/4
   198→    H_42 = expm(-(-F * eps * p_minus_x / 4) * T_z)  # = expm(+F*eps*p_minus_x/4 * T_z)
   199→
   200→    # Assemble in correct order (left-to-right = last-to-first applied)
   201→    # H = H_42 * H_cross_p_minus * H_41 * H_3 * H_22 * H_cross_p_plus * H_21 * H_1
   202→    H = H_42 @ H_cross_p_minus @ H_41 @ H_3 @ H_22 @ H_cross_p_plus @ H_21 @ H_1
   203→    return H

   206→# ----------------------------------------------------------------------
   207→# Numerical holonomy via parallel transport (segment-by-segment matrix exp)
   208→# ----------------------------------------------------------------------
   209→def stratified_holonomy_numerical_so3(x_c, y_c, eps, F, alpha_fn,
   210→                                       n_steps_per_edge=2001):
   211→    """Numerical SO(3) holonomy by parallel transport along the rectangular loop.
   212→
   213→    Discretizes each edge into n_steps_per_edge segments, computes the
   214→    so(3)-valued line integral A_mid * ds at each segment midpoint,
   215→    and accumulates H = expm(-A_mid * ds) * H_prev (left multiplication).
   216→
   217→    At boundary crossings (y1*y2 < 0), applies the SO(3) transition matrix
   218→    g_{+-}(x_cross)^{+/-1} (sign depends on crossing direction).
   219→
   220→    n_steps_per_edge MUST be odd so that no segment midpoint lands
   221→    exactly on y=0 (strict-sign-change crossing detection).
   222→    """
   223→    if n_steps_per_edge % 2 == 0:
   224→        n_steps_per_edge += 1
   225→
   226→    corners = [
   227→        (x_c - eps / 2, y_c - eps / 2),  # BL (bottom-left, in S_-)
   228→        (x_c + eps / 2, y_c - eps / 2),  # BR (bottom-right, in S_-)
   229→        (x_c + eps / 2, y_c + eps / 2),  # TR (top-right, in S_+)
   230→        (x_c - eps / 2, y_c + eps / 2),  # TL (top-left, in S_+)
   231→    ]
   232→    edges = []
   233→    for i in range(4):
   234→        start = corners[i]
   235→        end = corners[(i + 1) % 4]
   236→        ts = np.linspace(0.0, 1.0, n_steps_per_edge + 1)
   237→        pts = [(start[0] + t * (end[0] - start[0]),
   238→                start[1] + t * (end[1] - start[1])) for t in ts]
   239→        edges.append(pts)
   240→
   241→    H = I3.copy()  # accumulated holonomy (start at identity)
   242→    n_crossings = 0
   243→    for edge_idx, edge in enumerate(edges):
   244→        for i in range(len(edge) - 1):
   245→            x1, y1 = edge[i]
   246→            x2, y2 = edge[i + 1]
   247→            # Check boundary crossing (strict sign change of y)
   248→            if y1 * y2 < 0:
   249→                n_crossings += 1
   250→                # Find crossing point at y=0
   251→                t_cross = (0.0 - y1) / (y2 - y1)
   252→                x_cross = x1 + t_cross * (x2 - x1)
   253→                # Accumulate holonomy from edge[i] to crossing (in stratum of y1)
   254→                dx_seg1 = x_cross - x1
   255→                dy_seg1 = 0.0 - y1
   256→                x_mid1 = (x1 + x_cross) / 2
   257→                y_mid1 = (y1 + 0.0) / 2
   258→                A_seg1 = A_conn(x_mid1, y_mid1, dx_seg1, dy_seg1)
   259→                H = expm(-A_seg1) @ H
   260→                # Apply boundary transition: g(p)^{+/-1} (sign by direction)
   261→                if y1 > 0:
   262→                    # S_+ -> S_-: apply g_{+-}(p) = exp(+alpha(p) T_y)
   263→                    H = g_boundary(x_cross, alpha_fn) @ H
   264→                else:
   265→                    # S_- -> S_+: apply g_{+-}(p)^{-1} = exp(-alpha(p) T_y)
   266→                    H = g_boundary_inv(x_cross, alpha_fn) @ H
   267→                # Accumulate holonomy from crossing to edge[i+1] (in stratum of y2)
   268→                dx_seg2 = x2 - x_cross
   269→                dy_seg2 = y2 - 0.0
   270→                x_mid2 = (x_cross + x2) / 2
   271→                y_mid2 = (0.0 + y2) / 2
   272→                A_seg2 = A_conn(x_mid2, y_mid2, dx_seg2, dy_seg2)
   273→                H = expm(-A_seg2) @ H
   274→            else:
   275→                # No crossing: accumulate stratum holonomy
   276→                dx_seg = x2 - x1
   277→                dy_seg = y2 - y1
   278→                x_mid = (x1 + x2) / 2
   279→                y_mid = (y1 + y2) / 2
   280→                A_seg = A_conn(x_mid, y_mid, dx_seg, dy_seg)
   281→                H = expm(-A_seg) @ H
   282→    return H, n_crossings
